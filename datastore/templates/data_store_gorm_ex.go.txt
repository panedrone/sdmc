package dbal

import (
	"context"
	"os"

	//sdm:if pg
	"gorm.io/driver/postgres"
	//sdm:end
	//sdm:if mysql
	"gorm.io/driver/mysql"
	//sdm:end
	//sdm:if mssql
	"gorm.io/driver/sqlserver"
	//sdm:end
	//sdm:if sqlite
	"gorm.io/driver/sqlite" // cgo; without cgo: github.com/glebarez/sqlite
	//sdm:end
	"gorm.io/gorm"
	"gorm.io/gorm/logger"
)

func (ds *_DS) initDb() (err error) {
	dsn := os.Getenv("DATABASE_URL")
	//sdm:if pg
	if dsn == "" { // postgres://user:pwd@host:5432/db
		dsn = "postgres://postgres:postgres@localhost:5432/postgres?sslmode=disable"
	}
	dialector := postgres.Open(dsn)
	//sdm:end
	//sdm:if mysql
	if dsn == "" { // user:pwd@tcp(host:3306)/db?parseTime=true
		dsn = "root:root@tcp(localhost:3306)/mysql?parseTime=true" // parseTime: DATETIME -> time.Time
	}
	dialector := mysql.Open(dsn)
	//sdm:end
	//sdm:if mssql
	if dsn == "" { // sqlserver://user:pwd@host:1433?database=db, instance: host/SQLEXPRESS
		dsn = "sqlserver://sa:sa@localhost:1433?database=master"
	}
	dialector := sqlserver.Open(dsn)
	//sdm:end
	//sdm:if sqlite
	if dsn == "" { // path to the database file
		dsn = "./app.sqlite3"
	}
	dialector := sqlite.Open(dsn)
	//sdm:end
	ds.rootDb, err = gorm.Open(dialector, &gorm.Config{
		Logger: logger.Default.LogMode(logger.Warn),
	})
	return
}

var _ds = &_DS{}

func Ds() DataStore {
	return _ds
}

func init() {
	// generated DAO functions get DataStore via DS(ctx)
	DS = func(ctx context.Context) DataStore {
		return _ds
	}
}
