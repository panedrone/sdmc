class OutParam:
    """
    OUT and INOUT parameters of a procedure: pass it among the parameters, read value after the call.
    """

    def __init__(self, value=None):
        self.value = value
#sdm:end

def create_ds(conn) -> DataStore:
    """
    :param conn: a DB-API connection of {{driver}}
    """
    return _DS(conn)


class _DS(DataStore):

    def __init__(self, conn):
        self.conn = conn

    def in_transaction(self) -> bool:
#sdm:if sqlite
        return self.conn.in_transaction
#sdm:end
#sdm:if mysql
        return bool(self.conn.in_transaction)
#sdm:end
#sdm:if pg
        # libpq PQtransactionStatus, the same in psycopg2 (2.8+) and psycopg: 0 is IDLE
        return self.conn.info.transaction_status != 0
#sdm:end

    def begin(self):
#sdm:if sqlite
        self.conn.execute('begin')
#sdm:end
#sdm:if mysql
        self.conn.start_transaction()
#sdm:end
#sdm:if pg
        pass  # psycopg starts a transaction by itself with the first statement
#sdm:end

    def commit(self):
        self.conn.commit()

    def rollback(self):
        self.conn.rollback()

    def insert_row(self, sql, params, ai_values):
        sql = _format_sql(sql)
#sdm:if pg
        if len(ai_values) > 0:
            sql += ' RETURNING ' + ', '.join(ai[0] for ai in ai_values)
#sdm:end

        def do_insert(cursor):
            cursor.execute(sql, params)
            if cursor.rowcount == 0:
                raise Exception('No rows inserted')
            if len(ai_values) > 0:
#sdm:if pg
                row = cursor.fetchone()
                for i in range(len(ai_values)):
                    ai_values[i][1] = row[i]
#sdm:else
                ai_values[0][1] = cursor.lastrowid
#sdm:end

        self._exec(do_insert)

    def exec_dml(self, sql, params):
        sql = _format_sql(sql)
#sdm:if sqlite

        def do_exec(cursor):
            cursor.execute(sql, params)
            return cursor.rowcount

        return self._exec(do_exec)
#sdm:end
#sdm:if pg

        def do_exec(cursor):
            if _get_sp_name(sql) is None:
                cursor.execute(sql, params)
                return cursor.rowcount
            # a procedure: OUT and INOUT values come as the row of the call
            out_params = []
            call_params = []
            for p in params:
                if isinstance(p, OutParam):
                    call_params.append(p.value)
                    out_params.append(p)
                else:
                    call_params.append(p)
            cursor.execute(sql, call_params)
            if len(out_params) > 0:
                row0 = cursor.fetchone()
                for i, value in enumerate(row0):
                    out_params[i].value = value
            return 0

        return self._exec(do_exec)
#sdm:end
#sdm:if mysql
        sp = _get_sp_name(sql)

        def do_exec(cursor):
            if sp is None:
                cursor.execute(sql, params)
                return cursor.rowcount
            result_args = cursor.callproc(sp, _get_call_params(params))
            # the result sets of the procedure go to the callbacks of the DAO, in their order
            for p in params:
                if isinstance(p, list) and len(p) > 0 and callable(p[0]):
                    for i, result in enumerate(cursor.stored_results()):
                        _fetch_all(result, p[i])
                    break
            _assign_out_params(params, result_args)
            return 0

        return self._exec(do_exec)
#sdm:end

    def query_scalar(self, sql, params, out_params=None):
        rows = self.query_all_scalars(sql, params, out_params)
        if len(rows) == 0:
            raise Exception('No rows')
        if len(rows) > 1:
            raise Exception('More than 1 row exists')
        return rows[0]

    def query_all_scalars(self, sql, params, out_params=None):
        sql = _format_sql(sql)
        res = []
#sdm:if mysql
        sp = _get_sp_name(sql)
#sdm:end

        def fetch_all(cursor):
#sdm:if mysql
            if sp is not None:
                result_args = cursor.callproc(sp, _get_call_params(params))
                for result in cursor.stored_results():
                    for row in result:
                        res.append(row[0])
                _assign_out_params(params, result_args)
                return
#sdm:end
            cursor.execute(sql, params)
            for row in cursor:
                res.append(row[0])

        self._exec(fetch_all)
        return res

    def query_row(self, sql, params, out_params=None):
        rows = []
        self.query_all_rows(sql, params, lambda row: rows.append(row), out_params)
        if len(rows) == 1:
            return rows[0]
        if len(rows) == 0:
            return 'No rows'
        return 'More than 1 row exists'

    def query_all_rows(self, sql, params, callback, out_params=None):
        sql = _format_sql(sql)
#sdm:if mysql
        sp = _get_sp_name(sql)
#sdm:end

        def fetch_all(cursor):
#sdm:if mysql
            if sp is not None:
                result_args = cursor.callproc(sp, _get_call_params(params))
                for result in cursor.stored_results():
                    _fetch_all(result, callback)
                _assign_out_params(params, result_args)
                return
#sdm:end
            cursor.execute(sql, params)
            columns = [col[0] for col in cursor.description]
            for row in cursor:
                callback(dict(zip(columns, row)))

        self._exec(fetch_all)

    def _exec(self, func):
        cursor = self.conn.cursor()  # sqlite3 and mysql-connector cursors are not context managers
        try:
            return func(cursor)
        finally:
            cursor.close()


def _format_sql(sql):
#sdm:if sqlite
    return sql
#sdm:else
    return sql.replace('?', '%s')
#sdm:end
#sdm:if pg mysql


def _get_sp_name(sql):
    parts = sql.split()
    if len(parts) >= 2 and parts[0].strip().lower() == 'call':
        name = parts[1]
        end = name.find('(')
        if end == -1:
            return name
        return name[0:end]
    return None
#sdm:end
#sdm:if mysql


def _get_call_params(params):
    """
    mysql-connector: the values of IN, OUT and INOUT parameters in their order; the callbacks of the
    result sets are not parameters.
    """
    call_params = []
    for p in params:
        if isinstance(p, OutParam):
            call_params.append(p.value)
        elif isinstance(p, list) and len(p) > 0 and callable(p[0]):
            pass
        else:
            call_params.append(p)
    return call_params


def _assign_out_params(params, result_args):
    i = 0
    for p in params:
        if isinstance(p, list) and len(p) > 0 and callable(p[0]):
            continue  # not in result_args
        if isinstance(p, OutParam):
            p.value = result_args[i]
        i += 1


def _fetch_all(result, callback):
    # a result set of a procedure: cursor(dictionary=True) does not apply to it
    columns = [d[0] for d in result.description]
    for row in result:
        callback(dict(zip(columns, row)))
#sdm:end
